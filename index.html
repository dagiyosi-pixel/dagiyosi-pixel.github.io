vec3 parametric_torus(float u, float v) {
  float r = 1.25 + 0.38 * cos(v);
  float x = r * cos(u * 2.0);
  float y = r * sin(u * 2.0);
  float z = 0.38 * sin(v * 3.0);
  return vec3(x, y, z);
}
